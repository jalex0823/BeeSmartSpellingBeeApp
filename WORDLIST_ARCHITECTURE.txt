┌─────────────────────────────────────────────────────────────────────┐
│                         FLASK BACKEND                               │
│  ┌──────────────────────────────────────────────────────────────┐  │
│  │ Route: /api/saved-lists/load                                  │  │
│  │  • Loads words from database                                  │  │
│  │  • Sets session["active_list_id"] = list.id                   │  │
│  │  • Sets session["active_list_name"] = list.name               │  │
│  │  • Calls init_quiz_state()                                    │  │
│  └──────────────────────────────────────────────────────────────┘  │
│                                   │                                 │
│  ┌──────────────────────────────────────────────────────────────┐  │
│  │ Route: /quiz                                                   │  │
│  │  • Gets active_list_id from session (or generates hash)       │  │
│  │  • Creates selected_list = { id, name }                       │  │
│  │  • Passes to template                                         │  │
│  └──────────────────────────────────────────────────────────────┘  │
└─────────────────────────────────────────────────────────────────────┘
                                   │

                                   ▼
┌─────────────────────────────────────────────────────────────────────┐
│              WORD LIST MANAGER (quiz-wordlist.js)                   │
│  ┌──────────────────────────────────────────────────────────────┐  │
│  │ class WordListManager {                                       │  │
│  │   init(serverProvidedList) {                                  │  │
│  │     storedList = loadFromStorage()                            │  │
│  │     if (serverList.id !== storedList.id) {                    │  │
│  │       // Override storage with server list                    │  │
│  │       setActiveList(serverList)                               │  │
│  │     }                                                          │  │
│  │   }                                                            │  │
│  │                                                                │  │
│  │   setActiveList(listData) {                                   │  │
│  │     this.activeList = { id, name, words, ... }                │  │
│  │     saveToStorage()      // → localStorage                    │  │
│  │     notifyListChanged()  // → Update globals + dispatch event │  │
│  │   }                                                            │  │
│  │                                                                │  │
│  │   clearActiveList() {                                         │  │
│  │     localStorage.removeItem('beesmart_active_wordlist')       │  │
│  │     notifyListChanged()                                       │  │
│  │   }                                                            │  │
│  │ }                                                              │  │
│  └──────────────────────────────────────────────────────────────┘  │
└─────────────────────────────────────────────────────────────────────┘
                                   │
                    ┌──────────────┼──────────────┐
                    │              │              │
                    ▼              ▼              ▼
         ┌────────────────┐  ┌──────────┐  ┌────────────────┐
         │  localStorage  │  │ Globals  │  │ Custom Event   │
         │ ─────────────  │  │ ────────  │  │ ─────────────  │
         │ Key:           │  │ QUIZ_    │  │ wordlist:      │
         │ beesmart_      │  │ WORDS    │  │ changed        │
         │ active_        │  │          │  │                │
         │ wordlist       │  │ QUIZ_    │  │ detail: {      │
         │                │  │ CURRENT_ │  │   list,        │
         │ Value:         │  │ INDEX    │  │   words,       │
         │ {              │  │          │  │   listId,      │
         │   id,          │  │ QUIZ_    │  │   listName     │
         │   name,        │  │ ACTIVE_  │  │ }              │
         │   words,       │  │ LIST_ID  │  │                │
         │   loadedAt,    │  │          │  │                │
         │   version      │  │ QUIZ_    │  │                │
         │ }              │  │ ACTIVE_  │  │                │
         │                │  │ LIST_    │  │                │
         │                │  │ NAME     │  │                │
         └────────────────┘  └──────────┘  └────────────────┘
                    │              │              │
                    └──────────────┼──────────────┘

         └────────────────────────────────────────────────┘
